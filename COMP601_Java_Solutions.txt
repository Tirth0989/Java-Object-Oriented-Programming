		veges.add(new LeafyGreen(5, 2.94, "Lettuce"));
		veges.add(new LeafyGreen(20, 5.45, "Broccoli"));
		veges.add(new LeafyGreen(7, 1.94, "Lettuce"));
		veges.add(new LeafyGreen(15, 2.8, "Lettuce"));
		veges.add(new LeafyGreen(30, 2.5, "Cabbage"));
		veges.add(new LeafyGreen(28, 4.4, "Broccoli"));
		veges.add(new LeafyGreen(20, 5.5, "Cabbage"));
		veges.add(new LeafyGreen(16, 3.9, "Broccoli"));
		veges.add(new LeafyGreen(26, 4.8, "Broccoli"));

		// call method to calculate stats
		calculateStats(veges);
	}

	// method to calculate total cost and total weight of each type
	public static void calculateStats(List<LeafyGreen> veges) {
		double totalCost = 0.0;
		double cabbageWeight = 0.0;
		double broccoliWeight = 0.0;
		double lettuceWeight = 0.0;

		// loop through each vegetable in the list
		for (var lg : veges) {
			totalCost += lg.getCost();

			if (lg.getType().equalsIgnoreCase("Cabbage")) {
				cabbageWeight += lg.getWeight();
			} else if (lg.getType().equalsIgnoreCase("Broccoli")) {
				broccoliWeight += lg.getWeight();
			} else if (lg.getType().equalsIgnoreCase("Lettuce")) {
				lettuceWeight += lg.getWeight();
			}
		}

		// print the results
		println(String.format("Total vegetable cost: $%.1f", totalCost));
		println(String.format("Total Cabbage weight: %.1f kg", cabbageWeight));
		println(String.format("Total Broccoli weight: %.1f kg", broccoliWeight));
		println(String.format("Total Lettuce weight: %.1f kg", lettuceWeight));
	}
}

package Task7;

import module java.base;
import static java.lang.IO.*;

public class Main {

	void main() {

		// create a list of shapes
		var shapes = new LinkedList<Shape>();
		shapes.add(new Shape("white"));
		shapes.add(new Rectangle("red", 10, 6));
		shapes.add(new Rectangle("black", 20, 9));
		shapes.add(new Shape("orange"));

		// show shape names
		showShapeNames(shapes);

		// first array of rectangles
		Rectangle[] rectangleArray1 = {
				new Rectangle("white", 4, 3),
				new Rectangle("red", 9, 5),
				new Rectangle("purple", 3, 6),
				new Rectangle("orange", 15, 10),
				new Rectangle("black", 4, 14),
		};

		// second array of rectangles
		Rectangle[] rectangleArray2 = {
				new Rectangle("pink", 3, 4),
				new Rectangle("red", 10, 2),
				new Rectangle("white", 8, 5),
				new Rectangle("blue", 14, 4),
				new Rectangle("bindle", 10, 15),
		};

		// count overlapping rectangles
		countOverlapRectangles(rectangleArray1, rectangleArray2);
	}

	// print the type of each shape
	public static void showShapeNames(List<Shape> shapes) {
		for (var s : shapes) {
			println(s.getShapeType());
		}
	}

	// count overlapping colours and perimeters between two arrays
	public static void countOverlapRectangles(Rectangle[] group1, Rectangle[] group2) {
		var colours1 = new HashSet<String>();
		var perimeters1 = new HashSet<Double>();

		// store colour and perimeter from first array
		for (var r : group1) {
			colours1.add(r.getColour().trim().toLowerCase());
			perimeters1.add(r.getPerimeter());
		}

		int colourOverlap = 0;
		int perimeterOverlap = 0;

		// compare with second array
		for (var r : group2) {
			if (colours1.contains(r.getColour().trim().toLowerCase())) {
				colourOverlap++;
			}
			if (perimeters1.contains(r.getPerimeter())) {
				perimeterOverlap++;
			}
		}

		// print result
		println("There are " + colourOverlap + " Rectangle objects with overlapping colour between the two arrays");
		println("There are " + perimeterOverlap + " Rectangle objects with overlapping perimeter between the two arrays");
	}
}

package Task8;

import module java.base;
import static java.lang.IO.*;

public class BagApp {

	public double calcTotalCapacity(List<Bag> bagList) {
		// base case
		if (bagList == null || bagList.size() == 0) {
			return 0;
		}

		// recursive case
		var subList = bagList.subList(1, bagList.size());
		return bagList.get(0).getCapacity() + calcTotalCapacity(subList);
	}
}

package Task9;

import module java.base;
import static java.lang.IO.*;

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Paths;
import java.util.LinkedList;
import java.util.List;

public class GameApp {
	// list to store country objects
	private List<Country> countryList = new LinkedList<>();

	// constructor
	public GameApp(String filename) throws IOException {
		readData(filename);
	}

	// read file and store countries in list
	public void readData(String filename) throws IOException {
		var lines = Files.readAllLines(Paths.get(filename));

		for (var line : lines) {
			var items = line.split(",");

			String name = items[0].trim();
			String capital = items[1].trim();
			int population = Integer.parseInt(items[2].trim());

			countryList.add(new Country(name, capital, population));
		}
	}

	// guessing game method
	public void game() {
		if (countryList == null || countryList.size() == 0) {
			return;
		}

		var rand = new java.util.Random();
		int index = rand.nextInt(countryList.size());

		Country country = countryList.get(index);

		String guess = readln("What's the capital of " + country.getName() + "? ");

		if (guess.trim().equalsIgnoreCase(country.getCapital())) {
			println("Well done!");
		} else {
			println("Incorrect. It's " + country.getCapital() + ".");
		}

		println(country.getName() + " is a country of " + country.getPopulation() + " million.");
	}
}

package Task10;

import module java.base;
import static java.lang.IO.*;

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Paths;
import java.util.LinkedList;
import java.util.List;
import java.util.Random;

public class AudioBookApp {

	// list to store audiobook objects
	private List<AudioBook> bookList = new LinkedList<>();

	// getter for list
	public List<AudioBook> getBookList() {
		return bookList;
	}

	// constructor
	public AudioBookApp(String filename) throws IOException {
		readData(filename);
	}

	// read data from file
	public void readData(String filename) throws IOException {

		var lines = Files.readAllLines(Paths.get(filename));

		for (var line : lines) {

			var items = line.split(",");

			String name = items[0].trim();
			int year = Integer.parseInt(items[1].trim());
			double length = Double.parseDouble(items[2].trim());

			bookList.add(new AudioBook(name, year, length));
		}
	}

	// get shortest books
	public List<AudioBook> getShortestBooks() {

		var shortestBooks = new LinkedList<AudioBook>();

		if (bookList != null && bookList.size() > 0) {

			double shortest = bookList.get(0).getLength();

			// find shortest length
			for (var book : bookList) {

				if (book.getLength() < shortest) {
					shortest = book.getLength();
				}
			}

			// add shortest books into list
			for (var book : bookList) {

				if (book.getLength() == shortest) {
					shortestBooks.add(book);
				}
			}
		}

		return shortestBooks;
	}

	// count books between 2010 and 2020
	public int countBooks() {

		int count = 0;

		if (bookList != null && bookList.size() > 0) {

			for (var book : bookList) {

				if (book.getYear() >= 2010 &&
						book.getYear() <= 2020) {

					count++;
				}
			}
		}

		return count;
	}

	// divide books into random lists
	public void randomBookList() {

		var b1 = new LinkedList<AudioBook>();
		var b2 = new LinkedList<AudioBook>();
		var b3 = new LinkedList<AudioBook>();

		var rand = new Random();

		for (var book : bookList) {

			int num = rand.nextInt(3) + 1;

			if (num == 1) {
				b1.add(book);
			}

			else if (num == 2) {
				b2.add(book);
			}

			else {
				b3.add(book);
			}
		}

		// display list 1
		println(b1.size() + " books in list #1:");

		for (var book : b1) {
			book.displayInfo();
		}

		// display list 2
		println(b2.size() + " books in list #2:");

		for (var book : b2) {
			book.displayInfo();
		}

		// display list 3
		println(b3.size() + " books in list #3:");

		for (var book : b3) {
			book.displayInfo();
		}
	}
}
